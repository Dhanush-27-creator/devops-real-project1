pipeline {
    agent any

    stages {
        stage('Checkout') {
            steps {
                echo "Code checkout done"
            }
        }

        stage('Build Docker Image') {
            steps {
                sh 'docker build -t devops-demo-app:1.0 .'
            }
        }

        stage('Deploy App') {
            steps {
                sh '''
                docker rm -f devops-app || true
                docker run -d -p 3000:3000 --name devops-app devops-demo-app:1.0
                '''
            }
        }
    }
}
